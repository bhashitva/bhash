                byte[] bytes=new byte[buffer.remaining()];buffer.get(bytes);output.writeBytes(bytes);
            }

        }
        public void onError(Throwable error){result.completeExceptionally(error);}
        public void onComplete(){result.complete(output.toByteArray());}

    static long netSend(String method,String url,BhashRuntime.BList<String> headers,BhashRuntime.BBuffer body,long timeout){
        netArguments(method,url,headers,body,timeout);
        if(netResponses.size()>=32)throw failure("बहुत अधिक खुले HTTP उत्तर।");
        java.util.concurrent.CompletableFuture<java.net.http.HttpResponse<byte[]>> pending=null;
        long deadline=netClock()+timeout;
        try {
            var uri=java.net.URI.create(url);
            if(!List.of("http","https").contains(uri.getScheme())||uri.getRawUserInfo()!=null)throw failure("अमान्य HTTP पता।");
            byte[] bytes=new byte[(int)body.size()];for(int i=0;i<bytes.length;i++)bytes[i]=(byte)(short)body.get(i);
            var request=java.net.http.HttpRequest.newBuilder(uri).version(java.net.http.HttpClient.Version.HTTP_1_1)
                .timeout(Duration.ofMillis(timeout)).method(method,bytes.length==0?java.net.http.HttpRequest.BodyPublishers.noBody():java.net.http.HttpRequest.BodyPublishers.ofByteArray(bytes));
            for(int i=0;i<headers.size();i+=2)request.header(headers.get(i),headers.get(i+1));
            pending=NetClient.CLIENT.sendAsync(request.build(),info->{
                int count=0,size=0;
                for(var entry:info.headers().map().entrySet())for(String value:entry.getValue()){
                    size+=entry.getKey().length()+value.length()+4;
                    if(++count>256||size>NET_HEADERS)throw failure("HTTP शीर्षों की सीमा पार हो गई।");
                }
                return new NetSubscriber();
            });
            long remaining=deadline-netClock();if(remaining<=0)throw new java.util.concurrent.TimeoutException();
            var response=pending.get(remaining,java.util.concurrent.TimeUnit.MILLISECONDS);
            var result=new ArrayList<String>();
            for(var entry:response.headers().map().entrySet())for(String value:entry.getValue()){
                result.add(entry.getKey().toLowerCase(Locale.ROOT));result.add(value);
            }
            long id=++netNext;netResponses.put(id,new NetResponse(response.statusCode(),List.copyOf(result),response.body()));return id;
        }catch(java.util.concurrent.TimeoutException error){
            if(pending!=null)pending.cancel(true);throw failure("HTTP अनुरोध का समय समाप्त।");
        }catch(InterruptedException error){
            if(pending!=null)pending.cancel(true);Thread.currentThread().interrupt();throw failure("HTTP अनुरोध असफल।");
        }catch(java.util.concurrent.ExecutionException error){throw failure(netFailure(error));}
        catch(IllegalArgumentException error){throw failure("अमान्य HTTP पता।");}
    }

    private static java.sql.Connection dbConnect(String path)throws java.sql.SQLException {
        try{Class.forName("org.sqlite.JDBC");}
        catch(ClassNotFoundException|LinkageError missing){throw failure("SQLite रनटाइम उपलब्ध नहीं है।");}

        try{
            connection=java.sql.DriverManager.getConnection("jdbc:sqlite:"+path,properties);
            if(!connection.getMetaData().getDatabaseProductVersion().equals("3.54.1"))throw failure("SQLite रनटाइम संस्करण मेल नहीं खाता।");
