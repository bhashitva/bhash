import java.io.*;
import java.nio.charset.StandardCharsets;

/** Compiled into each generated program; no compiler or ANTLR dependency at runtime. */
final class BhashRuntime {
    private static final BufferedReader INPUT = new BufferedReader(new InputStreamReader(
            System.in, StandardCharsets.UTF_8.newDecoder()
                    .onMalformedInput(java.nio.charset.CodingErrorAction.REPORT)
                    .onUnmappableCharacter(java.nio.charset.CodingErrorAction.REPORT)));
    private static final PrintWriter OUTPUT = new PrintWriter(new OutputStreamWriter(System.out, StandardCharsets.UTF_8), true);
    private static final PrintWriter ERRORS = new PrintWriter(new OutputStreamWriter(System.err, StandardCharsets.UTF_8), true);

    static final class Failure extends RuntimeException {
        Failure(String message) { super(message); }
    }

    static long readInteger() {
        final String input;
        try { input = INPUT.readLine(); }
        catch (IOException error) { throw new Failure("कंसोल से UTF-8 इनपुट पढ़ा नहीं जा सका।"); }
        if (input == null) throw new Failure("इनपुट समाप्त: एक पूर्णांक अपेक्षित था।");
        String value = input.strip();
        if (!value.matches("[+-]?(?:[0-9]+|[०-९]+)"))
            throw new Failure("अमान्य इनपुट: एक पंक्ति में एक पूर्णांक लिखें (0-9 या ०-९)।");
        var ascii = new StringBuilder();
        for (char c : value.toCharArray()) ascii.append(c >= '०' && c <= '९' ? (char) ('0' + c - '०') : c);
        try { return Long.parseLong(ascii.toString()); }
        catch (NumberFormatException error) { throw new Failure("पूर्णांक 64-बिट सीमा से बाहर है।"); }
    }

    static String readText() {
        try {
            String value = INPUT.readLine();
            if (value == null) throw new Failure("इनपुट समाप्त: पाठ अपेक्षित था।");
            return value;
        } catch (IOException e) { throw new Failure("UTF-8 पाठ पढ़ा नहीं जा सका।"); }
    }
    static long length(String value) { return value.codePointCount(0, value.length()); }
    static String slice(String value, long start, long end) {
        long count = length(value);
        if (start < 0 || end < start || end > count) throw new Failure("पाठ अंश सीमा से बाहर है।");
        return value.substring(value.offsetByCodePoints(0, (int)start), value.offsetByCodePoints(0, (int)end));
    }
    static String integerText(long value) { return Long.toString(value); }
    static long parseInteger(String value) {
        if (!value.matches("[+-]?(?:[0-9]+|[०-९]+)")) throw new Failure("अमान्य पूर्णांक पाठ।");
        try { return Long.parseLong(value); }
        catch (NumberFormatException e) { throw new Failure("पूर्णांक 64-बिट सीमा से बाहर है।"); }
    }
    static String upper(String value) {
        var result = new StringBuilder();
        value.codePoints().forEach(cp -> result.appendCodePoint(cp >= 'a' && cp <= 'z' ? cp - 32 : cp));
        return result.toString();
    }
    static void fail(String message) { throw new Failure(message); }

    static void print(Object... values) {
        for (int i = 0; i < values.length; i++) {
            if (i > 0) OUTPUT.print(' ');
            Object value = values[i];
            if (value instanceof Long number) OUTPUT.print(devanagari(Long.toString(number)));
            else if (value instanceof Boolean flag) OUTPUT.print(flag ? "सत्य" : "असत्य");
            else if (value instanceof Double number) OUTPUT.print(devanagari(Double.toString(number)));
            else OUTPUT.print(value);
        }
        OUTPUT.println();
    }

    private static String devanagari(String ascii) {
        var result = new StringBuilder();
        for (char c : ascii.toCharArray()) result.append(c >= '0' && c <= '9' ? (char) ('०' + c - '0') : c);
        return result.toString();
    }

    // Mask a 64-bit Bhash count before narrowing it to the host shift-count type.
    static long shiftLeft(long value, long count) { return value << (int)(count & 63L); }
    static long shiftRight(long value, long count) { return value >> (int)(count & 63L); }
    static long shiftUnsigned(long value, long count) { return value >>> (int)(count & 63L); }

    static long divide(long left, long right) {
        if (right == 0) throw new Failure("शून्य से भाग देना संभव नहीं है।");
        return left / right;
    }
    static long remainder(long left, long right) {
        if (right == 0) throw new Failure("शून्य से शेषफल निकालना संभव नहीं है।");
        return left % right;
    }
    static boolean truth(boolean value) { return value; }
    static <T> T instance(T value) {
        if (value == null) throw new Failure("रिक्त वस्तु के सदस्य का उपयोग नहीं किया जा सकता।");
        return value;
    }
    static boolean same(Object left, Object right) { return left == right; }
    static void discard(Object value) {}
    static void report(String message) { ERRORS.println("भाष: " + message); }
}
