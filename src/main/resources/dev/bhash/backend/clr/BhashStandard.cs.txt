    private sealed class JsonReader {
        private readonly string source;private int at;
        internal JsonReader(string source){this.source=source;}
        private BhashRuntime.Failure invalid() => failure("अमान्य जेसन पाठ, स्थान "+at.ToString(CultureInfo.InvariantCulture)+"।");
        private void space(){while(at<source.Length&&" \t\r\n".IndexOf(source[at])>=0)at++;}
        private bool take(char c){if(at<source.Length&&source[at]==c){at++;return true;}return false;}
        private void need(char c){if(!take(c))throw invalid();}
        internal object read(){object result=value(0);space();if(at!=source.Length)throw invalid();return result;}
        private object value(int depth) {
            space();if(depth>64||at>=source.Length)throw invalid();char c=source[at];
            if(c=='"')return text();
            if(take('{')) {
                var result=BhashRuntime.typed(new BhashRuntime.BMap<string,object>(),"MAP:[String, Object]");space();if(take('}'))return result;
                while(true){space();if(at>=source.Length||source[at]!='"')throw invalid();string key=text();space();need(':');if(result.contains(key))throw invalid();result.set(key,value(depth+1));space();if(take('}'))return result;need(',');}
            }
            if(take('[')) {
                var result=BhashRuntime.typed(new BhashRuntime.BList<object>(),"LIST:[Object]");space();if(take(']'))return result;
                while(true){result.add(value(depth+1));space();if(take(']'))return result;need(',');}
            }
            foreach(string literal in new[]{"true","false","null"})if(source.AsSpan(at).StartsWith(literal.AsSpan(),StringComparison.Ordinal)){at+=literal.Length;return literal=="null"?null:(object)(literal=="true");}
            return number();
        }
        private string text() {
            need('"');var b=new StringBuilder();
            while(at<source.Length) {
                char c=source[at++];if(c=='"'){string s=b.ToString();unicode(s);return s;}if(c<32)throw invalid();
                if(c!='\\'){b.Append(c);continue;}if(at>=source.Length)throw invalid();c=source[at++];
                switch(c) {
                    case '"':case '\\':case '/':b.Append(c);break;
                    case 'b':b.Append('\b');break;case 'f':b.Append('\f');break;case 'n':b.Append('\n');break;case 'r':b.Append('\r');break;case 't':b.Append('\t');break;
                    case 'u':int code=0;for(int i=0;i<4;i++){if(at>=source.Length)throw invalid();char digit=source[at++];int n=digit>='0'&&digit<='9'?digit-'0':digit>='a'&&digit<='f'?digit-'a'+10:digit>='A'&&digit<='F'?digit-'A'+10:-1;if(n<0)throw invalid();code=code*16+n;}b.Append((char)code);break;
                    default:throw invalid();
                }
            }
            throw invalid();
        }
        private bool digit() => at<source.Length&&source[at]>='0'&&source[at]<='9';
        private object number() {
            int start=at;take('-');if(take('0')){if(digit())throw invalid();}else{if(!digit()||source[at]=='0')throw invalid();while(digit())at++;}
            bool real=false;
            if(take('.')){real=true;if(!digit())throw invalid();while(digit())at++;}
            if(take('e')||take('E')){real=true;if(!take('+'))take('-');if(!digit())throw invalid();while(digit())at++;}
            string s=source.Substring(start,at-start);
            if(!real){if(!long.TryParse(s,NumberStyles.AllowLeadingSign,CultureInfo.InvariantCulture,out long integer))throw invalid();return integer;}
            if(!double.TryParse(s,NumberStyles.Float,CultureInfo.InvariantCulture,out double value))throw invalid();return finite(value);
        }
    }
    internal static string stdJsonWrite(object value) {var writer=new JsonWriter();writer.value(value,0);return writer.output.ToString();}
    private sealed class JsonWriter {
        internal readonly StringBuilder output=new StringBuilder();
        private readonly HashSet<object> active=new HashSet<object>(ReferenceEqualityComparer.Instance);
        private void add(string s){if(output.Length>TEXT_LIMIT-s.Length)throw failure("जेसन पाठ सीमा से बाहर है।");output.Append(s);}
        private void text(string s) {
            unicode(s);add("\"");foreach(char c in s){if(c=='"'||c=='\\')add("\\"+c);else if(c<32)add("\\u"+((int)c).ToString("x4",CultureInfo.InvariantCulture));else add(c.ToString());}add("\"");
        }
        internal void value(object x,int depth) {
            if(depth>64)throw failure("जेसन की गहराई सीमा पार हो गई।");
            if(x==null){add("null");return;}if(x is string s){text(s);return;}if(x is bool b){add(b?"true":"false");return;}
            if(x is long||x is int||x is byte){add(Convert.ToString(x,CultureInfo.InvariantCulture));return;}
            if(x is double d){finite(d);string number=d.ToString("R",CultureInfo.InvariantCulture);if(!number.Contains('.')&&!number.Contains('E')&&!number.Contains('e'))number+=".0";add(number);return;}
            if(!active.Add(x))throw failure("चक्रीय संग्रह को जेसन में नहीं लिख सकते।");
            try {
                if(x is JsonSequence sequence){add("[");bool first=true;foreach(object item in sequence.jsonValues()){if(!first)add(",");first=false;value(item,depth+1);}add("]");return;}
                if(x is JsonObject map){add("{");bool first=true;foreach(var entry in map.jsonEntries()){if(!(entry.Key is string key))throw failure("जेसन मानचित्र की कुंजी पाठ होनी चाहिए।");if(!first)add(",");first=false;text(key);add(":");value(entry.Value,depth+1);}add("}");return;}
                throw failure("यह प्रकार जेसन में समर्थित नहीं है।");
            }finally{active.Remove(x);}
        }
    }
