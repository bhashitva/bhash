    private sealed class JsonReader {
        private readonly string source;private int at;
        internal JsonReader(string source){this.source=source;}
        private BhashRuntime.Failure invalid() => failure("अमान्य जेसन पाठ, स्थान "+at.ToString(CultureInfo.InvariantCulture)+"।");
        private void space(){while(at<source.Length&&" \t\r\n".IndexOf(source[at])>=0)at++;}
        private bool take(char c){if(at<source.Length&&source[at]==c){at++;return true;}return false;}
        private void need(char c){if(!take(c))throw invalid();}
        internal object read(){object result=value(0);space();if(at!=source.Length)throw invalid();return result;}
        private object value(int depth) {
            space();if(depth>64||at>=source.Length)throw invalid();char c=source[at];
            if(c=='"')return text();
            if(take('{')) {
                var result=BhashRuntime.typed(new BhashRuntime.BMap<string,object>(),"MAP:[String, Object]");space();if(take('}'))return result;
                while(true){space();if(at>=source.Length||source[at]!='"')throw invalid();string key=text();space();need(':');if(result.contains(key))throw invalid();result.set(key,value(depth+1));space();if(take('}'))return result;need(',');}
            }
            if(take('[')) {
                var result=BhashRuntime.typed(new BhashRuntime.BList<object>(),"LIST:[Object]");space();if(take(']'))return result;
                while(true){result.add(value(depth+1));space();if(take(']'))return result;need(',');}
            }
            foreach(string literal in new[]{"true","false","null"})if(source.AsSpan(at).StartsWith(literal.AsSpan(),StringComparison.Ordinal)){at+=literal.Length;return literal=="null"?null:(object)(literal=="true");}
            return number();
        }
        private string text() {
            need('"');var b=new StringBuilder();
            while(at<source.Length) {
                char c=source[at++];if(c=='"'){string s=b.ToString();unicode(s);return s;}if(c<32)throw invalid();
                if(c!='\\'){b.Append(c);continue;}if(at>=source.Length)throw invalid();c=source[at++];
                switch(c) {
                    case '"':case '\\':case '/':b.Append(c);break;
                    case 'b':b.Append('\b');break;case 'f':b.Append('\f');break;case 'n':b.Append('\n');break;case 'r':b.Append('\r');break;case 't':b.Append('\t');break;
                    case 'u':int code=0;for(int i=0;i<4;i++){if(at>=source.Length)throw invalid();char digit=source[at++];int n=digit>='0'&&digit<='9'?digit-'0':digit>='a'&&digit<='f'?digit-'a'+10:digit>='A'&&digit<='F'?digit-'A'+10:-1;if(n<0)throw invalid();code=code*16+n;}b.Append((char)code);break;
                    default:throw invalid();
                }
            }
            throw invalid();
        }
        private bool digit() => at<source.Length&&source[at]>='0'&&source[at]<='9';
        private object number() {
            int start=at;take('-');if(take('0')){if(digit())throw invalid();}else{if(!digit()||source[at]=='0')throw invalid();while(digit())at++;}
            bool real=false;
            if(take('.')){real=true;if(!digit())throw invalid();while(digit())at++;}
            if(take('e')||take('E')){real=true;if(!take('+'))take('-');if(!digit())throw invalid();while(digit())at++;}
            string s=source.Substring(start,at-start);
            if(!real){if(!long.TryParse(s,NumberStyles.AllowLeadingSign,CultureInfo.InvariantCulture,out long integer))throw invalid();return integer;}
            if(!double.TryParse(s,NumberStyles.Float,CultureInfo.InvariantCulture,out double value))throw invalid();return finite(value);
        }
    }

    private sealed class JsonWriter {
        internal readonly StringBuilder output=new StringBuilder();
        private readonly HashSet<object> active=new HashSet<object>(ReferenceEqualityComparer.Instance);
        private void add(string s){if(output.Length>TEXT_LIMIT-s.Length)throw failure("जेसन पाठ सीमा से बाहर है।");output.Append(s);}
        private void text(string s) {
            unicode(s);add("\"");foreach(char c in s){if(c=='"'||c=='\\')add("\\"+c);else if(c<32)add("\\u"+((int)c).ToString("x4",CultureInfo.InvariantCulture));else add(c.ToString());}add("\"");
        }
        internal void value(object x,int depth) {
            if(depth>64)throw failure("जेसन की गहराई सीमा पार हो गई।");
            if(x==null){add("null");return;}if(x is string s){text(s);return;}if(x is bool b){add(b?"true":"false");return;}
            if(x is long||x is int||x is byte){add(Convert.ToString(x,CultureInfo.InvariantCulture));return;}
            if(x is double d){finite(d);string number=d.ToString("R",CultureInfo.InvariantCulture);if(!number.Contains('.')&&!number.Contains('E')&&!number.Contains('e'))number+=".0";add(number);return;}
            if(!active.Add(x))throw failure("चक्रीय संग्रह को जेसन में नहीं लिख सकते।");
            try {
                if(x is JsonSequence sequence){add("[");bool first=true;foreach(object item in sequence.jsonValues()){if(!first)add(",");first=false;value(item,depth+1);}add("]");return;}
                if(x is JsonObject map){add("{");bool first=true;foreach(var entry in map.jsonEntries()){if(!(entry.Key is string key))throw failure("जेसन मानचित्र की कुंजी पाठ होनी चाहिए।");if(!first)add(",");first=false;text(key);add(":");value(entry.Value,depth+1);}add("}");return;}
                throw failure("यह प्रकार जेसन में समर्थित नहीं है।");
            }finally{active.Remove(x);}
        }
    }

    internal static long dbAsyncStart(long operation,long handle,string text,BhashRuntime.BList<object> values){
#if BHASH_BROWSER
        throw failure("ब्राउज़र लक्ष्य पर डेटाबेस उपलब्ध नहीं है।");

        if(code.Length!=5)code="XXXXX";foreach(char ch in code)if(!(ch>='A'&&ch<='Z'||ch>='0'&&ch<='9')){code="XXXXX";break;}
        return "PostgreSQL त्रुटि ["+code+"]।";
    }
    private static bool adDispatch(AdJob j){lock(j.Gate){if(j.Cancel.IsCancellationRequested)return false;j.Dispatched=true;return true;}}
    private static async System.Threading.Tasks.Task adPgCancel(AdJob j){
        var c=j.Connection;if(c==null||c.Closed)return;
        await adControlWorkers.WaitAsync().ConfigureAwait(false);
        try{
            IntPtr cancel=AdPg.CancelObject(c.Db);
            if(cancel!=IntPtr.Zero)await System.Threading.Tasks.Task.Run(()=>{try{AdPg.Cancel(cancel,new byte[256],256);}finally{AdPg.FreeCancel(cancel);}}).ConfigureAwait(false);
        }finally{adControlWorkers.Release();adClose(c);}
    }
    private static async System.Threading.Tasks.Task adPgCheck(AdJob j){
        bool timeout=netClock()>=j.Deadline;
        if(!timeout&&(!j.Cancel.IsCancellationRequested||j.Operation==7&&j.Dispatched))return;
        if(j.Operation==7&&j.Dispatched){adClose(j.Connection);throw new AdError("लेनदेन का परिणाम अज्ञात है; स्वचालित पुनःप्रयास न करें।");}
        await adPgCancel(j).ConfigureAwait(false);throw new AdError(timeout?"कार्य की समयसीमा समाप्त।":"कार्य निरस्त किया गया।");
    }
    private static async System.Threading.Tasks.Task adPgDrain(AdJob j,Func<IntPtr,bool> consume){
        var c=j.Connection;string error=null;
        for(;;){
            await adPgCheck(j).ConfigureAwait(false);
            int flush=AdPg.Flush(c.Db);if(flush<0||AdPg.Consume(c.Db)==0){adClose(c);throw new AdError(j.Operation==7?"लेनदेन का परिणाम अज्ञात है; स्वचालित पुनःप्रयास न करें।":adPgError(IntPtr.Zero));}
            if(flush==0&&AdPg.Busy(c.Db)==0){
                IntPtr result=AdPg.Result(c.Db);if(result==IntPtr.Zero)break;
                bool retained=false;
                try{
                    int status=PgNative.PQresultStatus(result);
                    if(status!=1&&status!=2&&status!=9)error=adPgError(result);
                    else if(error==null)retained=consume(result);
                }catch{
                    if(!retained)PgNative.PQclear(result);await adPgCancel(j).ConfigureAwait(false);throw;
                }
                if(!retained)PgNative.PQclear(result);
            }else await System.Threading.Tasks.Task.Delay(1).ConfigureAwait(false);
        }
        if(error!=null){if(c.Transaction)c.Failed=true;if(j.Operation==7&&(error.Contains("[08",StringComparison.Ordinal)||error.Contains("[XXXXX]",StringComparison.Ordinal))){adClose(c);throw new AdError("लेनदेन का परिणाम अज्ञात है; स्वचालित पुनःप्रयास न करें।");}throw new AdError(error);}
    }
    private static async System.Threading.Tasks.Task adPostgres(AdJob j){
        lock(adLoader){PgNative.Load();}
        if(j.Operation==1){
            string[] a=j.Text.Split('\n');if(a.Length!=10)throw new AdError("अमान्य PostgreSQL विन्यास।");
            string options="-c statement_timeout="+a[8]+" -c lock_timeout="+a[8]+" -c timezone=UTC -c datestyle=ISO -c standard_conforming_strings=on -c bytea_output=hex";
            string info="host="+pgQuote(a[0])+" port="+pgQuote(a[1])+" dbname="+pgQuote(a[2])+" user="+pgQuote(a[3])+" password="+pgQuote(a[4])+" sslmode="+pgQuote(a[5])+" connect_timeout="+pgQuote(a[7])+" client_encoding=UTF8 application_name=Bhash gssencmode=disable options="+pgQuote(options);
            if(a[6].Length!=0)info+=" sslrootcert="+pgQuote(a[6]);
            await adDataWorkers.WaitAsync().ConfigureAwait(false);
            try{await System.Threading.Tasks.Task.Run(()=>{if(!adDispatch(j))return;j.Opened=new AdConnection{Postgres=true,Db=AdPg.Start(adUtf8(info))};}).ConfigureAwait(false);}
            finally{adDataWorkers.Release();}
            if(j.Opened==null)return;var opened=j.Opened;
            if(opened.Db==IntPtr.Zero||PgNative.PQstatus(opened.Db)==1)throw new AdError(adPgError(IntPtr.Zero));
            int state=2;
            while(state!=3){
                if(j.Cancel.IsCancellationRequested||netClock()>=j.Deadline)throw new AdError("कार्य की समयसीमा समाप्त।");
                if(AdPg.Ready(opened.Db,state==2))state=AdPg.Connect(opened.Db);
                if(state==0)throw new AdError(adPgError(IntPtr.Zero));
                if(state!=3)await System.Threading.Tasks.Task.Delay(1).ConfigureAwait(false);
            }
            if(PgNative.PQserverVersion(opened.Db)<140000||AdPg.Nonblocking(opened.Db,1)!=0)throw new AdError(adPgError(IntPtr.Zero));return;
        }
        var c=j.Connection;if(!adDispatch(j))return;
        if(j.Operation==9){adClose(c);return;}
        if(j.Operation==5){if(j.Statement.Meta!=IntPtr.Zero)PgNative.PQclear(j.Statement.Meta);j.Statement.Meta=IntPtr.Zero;j.Statement.Closed=true;return;}
        if(c.Failed&&j.Operation!=8)throw new AdError("डेटालेनदेन असफल है; वापस लें।");
        if(j.Operation==2){
            if(j.Parameters.Length!=1||j.Parameters[0] is not string converted)throw new AdError("अमान्य PostgreSQL कथन।");
            var s=new AdStatement{Owner=c,Sql=adUtf8(converted)};j.Prepared=s;
            if(AdPg.PrepareQuery(c.Db,adUtf8(""),s.Sql,0,IntPtr.Zero)==0)throw new AdError(adPgError(IntPtr.Zero));
            await adPgDrain(j,_=>false).ConfigureAwait(false);
            if(AdPg.Describe(c.Db,adUtf8(""))==0)throw new AdError(adPgError(IntPtr.Zero));
            await adPgDrain(j,r=>{s.Meta=r;s.Parameters=PgNative.PQnparams(r);return true;}).ConfigureAwait(false);
            if(s.Parameters>999||PgNative.PQnfields(s.Meta)>256)throw new AdError("डेटा परिणाम की सीमा पार हो गई।");return;
        }
        if(j.Operation==3||j.Operation==4){
            var s=j.Statement;int columns=PgNative.PQnfields(s.Meta);
            if(j.Parameters.Length!=s.Parameters)throw new AdError("डेटाकथन के सभी मापदंड बाँधें।");
            if(j.Operation==3&&columns!=0||j.Operation==4&&columns==0)throw new AdError(j.Operation==3?"डेटाकथन पंक्तियाँ पढ़ने के लिए अगला उपयोग करें।":"डेटाकथन बदलाव के लिए चलाओ उपयोग करें।");
            var values=new IntPtr[j.Parameters.Length];
            try{
                for(int i=0;i<values.Length;i++){
                    object value=j.Parameters[i];if(value==null)continue;
                    string text=value switch{long n=>n.ToString(CultureInfo.InvariantCulture),double n=>n.ToString("G17",CultureInfo.InvariantCulture),bool flag=>flag?"true":"false",byte[] bytes=>"\\x"+Convert.ToHexString(bytes).ToLowerInvariant(),_ => (string)value};
                    if(text.IndexOf('\0')>=0)throw new AdError("अमान्य PostgreSQL पाठ।");
                    values[i]=System.Runtime.InteropServices.Marshal.StringToCoTaskMemUTF8(text);
                }
                if(AdPg.QueryParams(c.Db,s.Sql,values.Length,IntPtr.Zero,values,IntPtr.Zero,IntPtr.Zero,0)==0||AdPg.Single(c.Db)==0){adClose(c);throw new AdError(adPgError(IntPtr.Zero));}
            }finally{foreach(var value in values)if(value!=IntPtr.Zero)System.Runtime.InteropServices.Marshal.FreeCoTaskMem(value);}
            long bytesRead=0;
            await adPgDrain(j,r=>{
                int status=PgNative.PQresultStatus(r);
                if(j.Operation==4&&(status==9||status==2))adPgRows(j,r,ref bytesRead);
                else if(j.Operation==3&&status==1){string count=adString(PgNative.PQcmdTuples(r));j.Result=count.Length==0?0:long.Parse(count,CultureInfo.InvariantCulture);}
                return false;
            }).ConfigureAwait(false);
            if(j.Operation==4)j.Result=j.Rows.Count;return;
        }
        if(j.Operation==6&&c.Transaction)throw new AdError("डेटालेनदेन पहले सक्रिय है।");
        if(j.Operation==7&&!c.Transaction)throw new AdError("डेटालेनदेन सक्रिय नहीं है।");
        if(AdPg.Query(c.Db,adUtf8(j.Operation==6?"BEGIN":j.Operation==7?"COMMIT":"ROLLBACK"))==0){adClose(c);throw new AdError(j.Operation==7?"लेनदेन का परिणाम अज्ञात है; स्वचालित पुनःप्रयास न करें।":adPgError(IntPtr.Zero));}
        await adPgDrain(j,_=>false).ConfigureAwait(false);c.Transaction=j.Operation==6;c.Failed=false;
    }
    private static void adPgRows(AdJob j,IntPtr result,ref long total){
        int columns=PgNative.PQnfields(result);if(columns>256)throw new AdError("डेटा परिणाम की सीमा पार हो गई।");
        if(j.Names.Length==0){j.Names=new string[columns];for(int i=0;i<columns;i++)j.Names[i]=adString(PgNative.PQfname(result,i));}
        if(PgNative.PQntuples(result)==0)return;
        if(j.Rows.Count>=1024)throw new AdError("डेटा परिणाम की सीमा १०२४ पंक्तियाँ / ८३८८६०८ बाइट है।");
        var row=new AdCell[columns];
        for(int i=0;i<columns;i++){
            if(PgNative.PQgetisnull(result,0,i)!=0){total+=8;if(total>DB_LIMIT)throw new AdError("डेटा परिणाम की सीमा १०२४ पंक्तियाँ / ८३८८६०८ बाइट है।");row[i]=new AdCell("रिक्त",null);continue;}
            string kind=pgKind(PgNative.PQftype(result,i));int length=PgNative.PQgetlength(result,0,i);
            long bytes=kind=="बाइटबफ़र"?(length-2)/2:kind=="पूर्णांक"||kind=="दशमलव"||kind=="तार्किक"?8:length;
            if(bytes<0||bytes>DB_LIMIT-total)throw new AdError("डेटा परिणाम की सीमा १०२४ पंक्तियाँ / ८३८८६०८ बाइट है।");total+=bytes;
            byte[] raw=new byte[length];if(length!=0)System.Runtime.InteropServices.Marshal.Copy(PgNative.PQgetvalue(result,0,i),raw,0,length);
            string text=new UTF8Encoding(false,true).GetString(raw);
            object value=kind switch{
                "पूर्णांक"=>long.Parse(text,CultureInfo.InvariantCulture),"दशमलव"=>double.Parse(text,CultureInfo.InvariantCulture),"तार्किक"=>text=="t",
                "बाइटबफ़र"=>text.StartsWith("\\x",StringComparison.Ordinal)?Convert.FromHexString(text.Substring(2)):throw new AdError("अमान्य PostgreSQL बाइटबफ़र।"),
                "पाठ" or "समय" or "सटीकदशमलव"=>text,_=>throw new AdError("डेटाकथन स्तंभ का प्रकार मेल नहीं खाता।")};
            if(value is double real&&!double.IsFinite(real))throw new AdError("डेटाकथन में सीमित दशमलव मान दें।");row[i]=new AdCell(kind,value);
        }
        j.Rows.Add(row);
    }
