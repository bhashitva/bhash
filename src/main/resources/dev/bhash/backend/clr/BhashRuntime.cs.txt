    internal static long readInteger() {
        string input;
        try { input = Input.ReadLine(); }
        catch (Exception error) when (error is IOException || error is DecoderFallbackException) {
            throw new Failure("कंसोल से UTF-8 इनपुट पढ़ा नहीं जा सका।");
        }
        if (input == null) throw new Failure("इनपुट समाप्त: एक पूर्णांक अपेक्षित था।");
        int start = 0, end = input.Length;
        while (start < end && whitespace(input[start])) start++;
        while (end > start && whitespace(input[end - 1])) end--;
        string value = input.Substring(start, end - start);
        if (!Regex.IsMatch(value, @"\A[+-]?(?:[0-9]+|[०-९]+)\z", RegexOptions.CultureInvariant))
            throw new Failure("अमान्य इनपुट: एक पंक्ति में एक पूर्णांक लिखें (0-9 या ०-९)।");
        var ascii = new StringBuilder();
        foreach (char c in value) ascii.Append(c >= '०' && c <= '९' ? (char)('0' + c - '०') : c);
        if (!long.TryParse(ascii.ToString(), NumberStyles.AllowLeadingSign, CultureInfo.InvariantCulture, out long number))
            throw new Failure("पूर्णांक 64-बिट सीमा से बाहर है।");
        return number;
    }

    // Match Java Character.isWhitespace / String.strip, including its nonbreaking-space exclusions.
    private static bool whitespace(char c) => (c >= '\t' && c <= '\r') || (c >= '\u001c' && c <= ' ')
        || c == '\u1680' || (c >= '\u2000' && c <= '\u2006') || (c >= '\u2008' && c <= '\u200a')
        || c == '\u2028' || c == '\u2029' || c == '\u205f' || c == '\u3000';

    }
    internal static long length(string value) {
        long count = 0; foreach (var rune in value.EnumerateRunes()) count++; return count;
    }
    internal static string slice(string value, long start, long end) {
        if (start < 0 || end < start || end > length(value)) throw new Failure("पाठ अंश सीमा से बाहर है।");
        var result = new StringBuilder(); long i = 0;
        foreach (var rune in value.EnumerateRunes()) { if (i >= start && i < end) result.Append(rune.ToString()); i++; }
        return result.ToString();
    }

        if (!Regex.IsMatch(value, @"\A[+-]?(?:[0-9]+|[०-९]+)\z", RegexOptions.CultureInvariant)) throw new Failure("अमान्य पूर्णांक पाठ।");

    internal static string upper(string value) {
        var result = new StringBuilder();
        foreach (var rune in value.EnumerateRunes()) {
            int cp = rune.Value; result.Append(char.ConvertFromUtf32(cp >= 'a' && cp <= 'z' ? cp - 32 : cp));
        }
        return result.ToString();
    }

    internal static void print(params object[] values) {
        for (int i = 0; i < values.Length; i++) {
            if (i > 0) Output.Write(' ');
            object value = values[i];
            if (value is long number) Output.Write(devanagari(number.ToString(CultureInfo.InvariantCulture)));
            else if (value is int || value is byte) Output.Write(devanagari(Convert.ToString(value,CultureInfo.InvariantCulture)));
            else if (value is bool flag) Output.Write(flag ? "सत्य" : "असत्य");
            else if (value is double real) Output.Write(devanagari(realText(real)));
            else Output.Write(value);
        }
        Output.WriteLine();
    }

    private static string devanagari(string ascii) {
        var result = new StringBuilder();
        foreach (char c in ascii) result.Append(c >= '0' && c <= '9' ? (char)('०' + c - '0') : c);
        return result.ToString();
    }

    // Normalize .NET round-trip formatting to Bhash's decimal/exponent presentation.
    // Some shortest representations still differ from Java 17 in their last digits.
    private static string realText(double number) {
        if (double.IsNaN(number)) return "NaN";
        if (double.IsPositiveInfinity(number)) return "Infinity";
        if (double.IsNegativeInfinity(number)) return "-Infinity";
        if (number == 0) return BitConverter.DoubleToInt64Bits(number) < 0 ? "-0.0" : "0.0";
        string raw = Math.Abs(number).ToString("R", CultureInfo.InvariantCulture);
        int exponent = 0, e = raw.IndexOf('E');
        if (e >= 0) { exponent = int.Parse(raw.Substring(e + 1), CultureInfo.InvariantCulture); raw = raw.Substring(0, e); }
        int dot = raw.IndexOf('.');
        int position = (dot < 0 ? raw.Length : dot) + exponent;
        string digits = raw.Replace(".", "");
        while (digits.Length > 1 && digits[0] == '0') { digits = digits.Substring(1); position--; }
        digits = digits.TrimEnd('0');
        int power = position - 1;
        string result;
        if (power >= 7 || power < -3)
            result = digits[0] + "." + (digits.Length > 1 ? digits.Substring(1) : "0") + "E" + power.ToString(CultureInfo.InvariantCulture);
        else if (position <= 0) result = "0." + new string('0', -position) + digits;
        else if (position >= digits.Length) result = digits + new string('0', position - digits.Length) + ".0";
        else result = digits.Insert(position, ".");
        return number < 0 ? "-" + result : result;
    }

    internal static string text(int[] scalars) {
        var result = new StringBuilder();

    // Explicit unsigned reinterpretation gives zero-fill right shifts on every CLR target.
    internal static long shiftLeft(long value, long count) => unchecked(value << (int)(count & 63L));
    internal static long shiftRight(long value, long count) => value >> (int)(count & 63L);
    internal static long shiftUnsigned(long value, long count) => unchecked((long)((ulong)value >> (int)(count & 63L)));

    internal static long divide(long left, long right) {
        if (right == 0) throw new Failure("शून्य से भाग देना संभव नहीं है।");
        return left == long.MinValue && right == -1 ? long.MinValue : left / right;
    }
    internal static long remainder(long left, long right) {
        if (right == 0) throw new Failure("शून्य से शेषफल निकालना संभव नहीं है।");
        return left == long.MinValue && right == -1 ? 0 : left % right;
    }

    private static int collectionIndex(long index,int size,bool end) {

    private static void collectionVersion(long expected,long actual) {
        if (expected!=actual) throw new Failure("पुनरावृत्ति के दौरान संग्रह का आकार बदला गया।");
    }
    private static bool collectionSame(object a,object b) {
        if (ReferenceEquals(a,b)) return true;
        if (a is double x && b is double y) return x==y || double.IsNaN(x) && double.IsNaN(y);
        if (a is string text && b is string other) return string.Equals(text,other,StringComparison.Ordinal);
        if (a is long || a is int || a is byte || a is bool || a is BRecord) return a.Equals(b);
        return false;
    }
    private static int collectionHash(object value) {
        if (value is double d) {
            long bits=d==0.0?0L:double.IsNaN(d)?0x7ff8000000000000L:BitConverter.DoubleToInt64Bits(d);
            return unchecked((int)(bits^(bits>>32)));
        }
        if (value is string text) { int hash=0; foreach(char c in text) hash=unchecked(31*hash+c); return hash; }
        if (value is long number) return unchecked((int)(number^(number>>32)));
        if (value is bool flag) return flag?1231:1237;
        if (value is int || value is byte || value is BRecord) return value.GetHashCode();
        return System.Runtime.CompilerServices.RuntimeHelpers.GetHashCode(value);
    }
    private sealed class BKey<T> {
        internal readonly T value;
        internal BKey(T value) { if ((object)value==null) throw new Failure("मानचित्र की कुंजी रिक्त नहीं हो सकती।"); this.value=value; }
        public override bool Equals(object other) => other is BKey<T> key && collectionSame(value,key.value);
        public override int GetHashCode() => collectionHash(value);
    }
    internal abstract class BSeq<T> : BTyped, System.Collections.Generic.IEnumerable<T> {
        internal abstract long size();
        internal abstract T get(long index);
        internal abstract void set(long index,T value);
        internal virtual long version() => 0;
        internal bool empty() => size()==0;
        internal bool contains(T value) { foreach (T item in this) if (collectionSame(item,value)) return true; return false; }
        public System.Collections.Generic.IEnumerator<T> GetEnumerator() {
            long expected=version(),position=0;
            while (true) {
                collectionVersion(expected,version());
                if (position>=size()) yield break;
                yield return get(position++);
            }
        }
        System.Collections.IEnumerator System.Collections.IEnumerable.GetEnumerator() => GetEnumerator();
    }

    }
    internal sealed class BList<T> : BSeq<T> {
        private readonly System.Collections.Generic.List<T> items=new System.Collections.Generic.List<T>();
        private long stamp;
        internal override long version() => stamp;
        internal override long size() => items.Count;
        internal override T get(long index) => items[collectionIndex(index,items.Count,false)];
        internal override void set(long index,T value) { items[collectionIndex(index,items.Count,false)]=value; }
        internal void add(T value) { items.Add(value); stamp++; }
        internal void insert(long index,T value) { items.Insert(collectionIndex(index,items.Count,true),value); stamp++; }
        internal T remove(long index) { int at=collectionIndex(index,items.Count,false); T value=items[at]; items.RemoveAt(at); stamp++; return value; }

    }
    internal sealed class BEntry<K,V> {
        internal readonly K key; internal readonly V value;

    internal sealed class BMap<K,V> : BTyped, System.Collections.Generic.IEnumerable<K> {
        private sealed class Cell {
            internal V value; internal readonly System.Collections.Generic.LinkedListNode<BKey<K>> node;
            internal Cell(V value,System.Collections.Generic.LinkedListNode<BKey<K>> node) { this.value=value; this.node=node; }
        }
        private readonly System.Collections.Generic.Dictionary<BKey<K>,Cell> items=new System.Collections.Generic.Dictionary<BKey<K>,Cell>();
        private readonly System.Collections.Generic.LinkedList<BKey<K>> order=new System.Collections.Generic.LinkedList<BKey<K>>();
        private long stamp;
        internal long size() => items.Count;
        internal bool empty() => items.Count==0;
        internal bool contains(K key) => items.ContainsKey(new BKey<K>(key));
        internal V get(K key) {
            if (!items.TryGetValue(new BKey<K>(key),out Cell cell)) throw new Failure("मानचित्र में कुंजी नहीं मिली।");
            return cell.value;
        }
        internal V getOr(K key,V fallback) => items.TryGetValue(new BKey<K>(key),out Cell cell)?cell.value:fallback;
        internal void set(K key,V value) {
            var wrapped=new BKey<K>(key);
            if (items.TryGetValue(wrapped,out Cell cell)) { cell.value=value; return; }
            var node=order.AddLast(wrapped); items.Add(wrapped,new Cell(value,node)); stamp++;
        }
        internal bool remove(K key) {
            var wrapped=new BKey<K>(key);
            if (!items.TryGetValue(wrapped,out Cell cell)) return false;
            items.Remove(wrapped); order.Remove(cell.node); stamp++; return true;
        }
        internal void clear() { if (items.Count!=0) { items.Clear(); order.Clear(); stamp++; } }
        internal System.Collections.Generic.IEnumerable<BEntry<K,V>> entries() {
            long expected=stamp; var node=order.First;
            while (true) {
                collectionVersion(expected,stamp);
                if (node==null) yield break;
                var key=node.Value; node=node.Next;
                yield return new BEntry<K,V>(key.value,items[key].value);
            }
        }
        public System.Collections.Generic.IEnumerator<K> GetEnumerator() { foreach (var entry in entries()) yield return entry.key; }
        System.Collections.IEnumerator System.Collections.IEnumerable.GetEnumerator() => GetEnumerator();
        internal BList<K> keys() { var result=new BList<K>(); foreach(K key in this) result.add(key); return result; }
        internal BList<V> values() { var result=new BList<V>(); foreach(var entry in entries()) result.add(entry.value); return result; }
    }

    internal static long toLong(double value) {
        if (!double.IsFinite(value) || value < -9223372036854775808.0 || value >= 9223372036854775808.0) throw new Failure("संख्यात्मक परिवर्तन सीमा से बाहर है।");

    }

    internal static double parseReal(string value) {
        if (!Regex.IsMatch(value,@"\A[+-]?(?:[0-9]+(?:\.[0-9]+)?|[०-९]+(?:\.[०-९]+)?)\z",RegexOptions.CultureInvariant)) throw new Failure("अमान्य दशमलव पाठ।");
        var ascii=new StringBuilder(); foreach(char c in value) ascii.Append(c>='०'&&c<='९'?(char)('0'+c-'०'):c);
        if (!double.TryParse(ascii.ToString(),NumberStyles.AllowLeadingSign|NumberStyles.AllowDecimalPoint,CultureInfo.InvariantCulture,out double result) || !double.IsFinite(result))
            throw new Failure("दशमलव परिवर्तन सीमा से बाहर है।");
        return result;
    }

    internal class BTyped { internal string typeTag; }

    internal static bool isType(object value,string tag,Type type) => value!=null && type.IsInstanceOfType(value) && (tag.Length==0 || value is BTyped item && item.typeTag==tag);
    internal static T checkedCast<T>(object value,string tag,Type type,bool nullable) {
        if(value==null && nullable) return default(T);
        if(!isType(value,tag,type)) throw new Failure("रनटाइम प्रकार परिवर्तन असफल हुआ।");

    }
    internal static T safeCast<T>(object value,string tag,Type type) => isType(value,tag,type)?(T)value:default(T);
    public abstract class BRecord {
        protected abstract object[] dataFields();
        public sealed override bool Equals(object other) {
            if(other==null || other.GetType()!=GetType()) return false;
            object[] a=dataFields(),b=((BRecord)other).dataFields();
            for(int i=0;i<a.Length;i++) if(!collectionSame(a[i],b[i])) return false;
            return true;
        }
        public sealed override int GetHashCode() { int hash=GetType().GetHashCode(); foreach(object item in dataFields()) hash=unchecked(31*hash+collectionHash(item)); return hash; }
    }
    internal sealed class BBuffer : BSeq<byte> {
        internal readonly byte[] items;
        internal BBuffer(long size) { items=new byte[collectionLength(size)]; }
        internal BBuffer(byte[] data) { items=data; }
        internal override long size() => items.Length;
        internal override byte get(long index) => items[collectionIndex(index,items.Length,false)];
        internal override void set(long index,byte value) { items[collectionIndex(index,items.Length,false)]=value; }
        private int span(long offset) {
            if(offset<0 || offset>items.Length-8L) throw new Failure("बाइटबफ़र सीमा से बाहर है।");
            return (int)offset;
        }
        internal long readLong(long offset,bool big) {
            int at=span(offset); ulong result=0;
            for(int i=0;i<8;i++) result|=(ulong)items[at+i] << (8*(big?7-i:i));
            return unchecked((long)result);
        }
        internal void writeLong(long offset,long value,bool big) {
            int at=span(offset); for(int i=0;i<8;i++) items[at+i]=unchecked((byte)((ulong)value >> (8*(big?7-i:i))));
        }
        internal double readReal(long offset,bool big) => BitConverter.Int64BitsToDouble(readLong(offset,big));
        internal void writeReal(long offset,double value,bool big) { writeLong(offset,BitConverter.DoubleToInt64Bits(value),big); }
        internal void fill(byte value) { Array.Fill(items,value); }
        internal BBuffer slice(long start,long end) {
            if(start<0 || end<start || end>items.Length) throw new Failure("बाइटबफ़र सीमा से बाहर है।");
            var result=new byte[(int)(end-start)]; Array.Copy(items,(int)start,result,0,result.Length); return new BBuffer(result);
        }
        internal BArray<byte> toArray() => new BArray<byte>(items);
    }
    internal static BBuffer bufferOf(params byte[] values) { var result=new BBuffer(values.LongLength); for(int i=0;i<values.Length;i++) result.set(i,values[i]); return result; }
